The Versat architecture originated the following publications; if this
core is used in derived work, please consider citing them.

\begin{itemize}
  \itemsep-0.5em
  \item J. D. Lopes and J. T. de Sousa, ``Versat, a Minimal Coarse-Grain
    Reconfigurable Array,'' \emph{12th International Meeting on
    High-Performance Computing for Computational Science (VECPAR)}, 2016.
  \item J. D. Lopes, R. Santiago, and J. T. de Sousa, ``Versat, a Runtime
    Partially Reconfigurable Coarse-Grain Reconfigurable Array using a
    Programmable Controller,'' \emph{Jornadas Sarteco}, pp. 561--569, 2016.
  \item R. Santiago, J. D. Lopes, and J. T. de Sousa, ``Compiler for the
    Versat Reconfigurable Architecture,'' \emph{Jornadas sobre Sistemas
    Reconfigur\'aveis (REC)}, pp. 41--48, 2017.
  \item J. D. Lopes and J. T. de Sousa, ``Fast Fourier Transform on the
    Versat CGRA,'' \emph{Jornadas Sarteco}, 2017.
  \item J. D. Lopes, J. T. de Sousa, H. Neto, and M. V\'estias, ``K-means
    clustering on CGRA,'' \emph{27th Int. Conference on Field Programmable
    Logic and Applications}, 2017.
  \item L. Fiolhais, F. Gon\c{c}alves, R. P. Duarte, M. V\'estias, and J. T.
    de Sousa, ``Low energy heterogeneous computing with multiple RISC-V and
    CGRA cores,'' \emph{2019 IEEE International Symposium on Circuits and
    Systems (ISCAS)}, pp. 1--5, 2019.
  \item V. M\'ario, J. D. Lopes, M. V\'estias, and J. T. de Sousa,
    ``Implementing CNNs using a linear array of full mesh CGRAs,''
    \emph{Applied Reconfigurable Computing (ARC)}, pp. 288--297, 2020.
  \item J. D. Lopes, M. P. V\'estias, R. P. Duarte, H. C. Neto, and J. T. de
    Sousa, ``Coarse-Grained Reconfigurable Computing with the Versat
    Architecture,'' \emph{Electronics}, vol. 10, no. 6, 2021.
  \item R. Teixeira, J. B. Rodrigues, J. Aguiar, J. Li, and J. T. de Sousa,
    ``Versat-AI: An ONNX-to-SoC Compiler for Model-Agnostic CGRA Edge
    Inference,'' \emph{Journal of Low Power Electronics and Applications},
    vol. 16, no. 3, art. 36, 2026.
\end{itemize}
